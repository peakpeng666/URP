canon=lambda x:json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False)
sha=lambda x:hashlib.sha256(x).hexdigest()
def put(p,x):
    with p.open('x') as f:f.write(canon(x)+'\n')
def seed(*parts):return int.from_bytes(hashlib.sha256((NS+'|'+'|'.join(map(str,parts))).encode()).digest()[:16],'big')
def make(family,n,rng):
    if family=='weibull_3_45':a=45*rng.weibull(3,n)
    elif family=='uniform_1_100':a=rng.integers(1,101,n)
    elif family=='small_beta_2_5':a=100*rng.beta(2,5,n)
    elif family=='large_beta_5_2':a=100*rng.beta(5,2,n)
    elif family=='normal_50_15':a=rng.normal(50,15,n)
    else:a=np.where(rng.integers(0,2,n)==0,rng.integers(20,31,n),rng.integers(70,81,n))
    return np.rint(np.clip(a,1,100)).astype(np.int64).tolist(),{'raw_below_1':int(np.sum(a<1)),'raw_above_100':int(np.sum(a>100))}
records=[]
for split,reps in [('train',8),('confirmation',6)]:
    for family in FAMILIES:
        for rep in range(reps):
            v_seed=seed(split,family,rep,'values'); o_seed=seed(split,family,rep,'permutation')
            items,clips=make(family,5000,np.random.Generator(np.random.PCG64(v_seed)))
            multiset_sha=sha(canon(sorted(items)).encode()); group=f'{split}_{family}_r{rep}'
            perm=np.random.Generator(np.random.PCG64(o_seed)).permutation(5000).tolist()
            for order in ORDERS:
                idx=perm if order=='random' else sorted(range(5000),key=lambda j:items[j],reverse=order=='descending')
                meta={'instance_id':f'{group}__{order}','group_id':group,'split':split,'family':family,'replicate':rep,'order':order,'value_seed':v_seed,'permutation_seed':o_seed,'multiset_sha256':multiset_sha,**clips}
                stream=[items[j] for j in idx]
                put(H/'inputs'/f"{meta['instance_id']}.json",{**meta,'capacity':100,'n':5000,'items':stream,'source_position':idx})
                records.append(meta)
# Risk panel: four random 500-prefixes, each followed by the same 4500-item suffix
# multiset in random/ascending/descending arrival order. Not used in training or main metric.

    p_seed=seed('risk',family,'prefix'); s_seed=seed('risk',family,'suffix_permutation')
    prefix,clips=make(family,500,np.random.Generator(np.random.PCG64(p_seed)))
    suffix=list(range(1,101))*45; perm=np.random.Generator(np.random.PCG64(s_seed)).permutation(4500).tolist()
    group=f'risk_prefix_{family}'; prefix_sha=sha(canon(prefix).encode())
    for order in ORDERS:
        tail=[suffix[j] for j in perm] if order=='random' else sorted(suffix,reverse=order=='descending')
        stream=prefix+tail
        meta={'instance_id':f'{group}__{order}','group_id':group,'split':'risk','family':family,'replicate':rep,'order':order,'prefix_seed':p_seed,'suffix_permutation_seed':s_seed,'prefix_sha256':prefix_sha,'multiset_sha256':sha(canon(sorted(stream)).encode()),**clips}
        put(H/'inputs'/f"{meta['instance_id']}.json",{**meta,'capacity':100,'n':5000,'items':stream})
        records.append(meta)
