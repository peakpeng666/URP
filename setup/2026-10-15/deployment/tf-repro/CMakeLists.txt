cmake_minimum_required(VERSION 3.16)
project(urp_tf_repro)
find_package(ament_cmake REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rclcpp REQUIRED)
add_executable(tf-repro main.cpp)
ament_target_dependencies(tf-repro tf2_ros rclcpp)
target_compile_features(tf-repro PRIVATE cxx_std_17)
add_library(urp_tf_buffer SHARED buffer_candidate.cpp)
ament_target_dependencies(urp_tf_buffer tf2_ros rclcpp)
target_compile_features(urp_tf_buffer PRIVATE cxx_std_17)
add_executable(tf-semantics semantics.cpp)
ament_target_dependencies(tf-semantics tf2_ros rclcpp)
target_compile_features(tf-semantics PRIVATE cxx_std_17)
target_compile_options(tf-semantics PRIVATE -UNDEBUG)
